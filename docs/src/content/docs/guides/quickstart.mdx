## Machine Demo


This code shows how to:
- Create a `machine` that manage a promise
- Subscribe its state with `useMachine`
- Conditionally render React components `state.match(cases)`
- Render the machine `state.key` and `state.data`
- Render the last change using `machine.getChange()`
- Trigger an event using `machine.send(type, ...args)`
- Trigger an event using `machine.event[type](...args)`
- Reset the machine using `machine.reset()`

### Code 

```tsx twoslash
import { createPromiseMachine, withSubscribe, useMachine } from "./src/matchina-source-lib";
// import { createPromiseMachine, withSubscribe } from "matchina";
// import { useMachine } from "matchina/extras/react";
// import { useMachine } from "../../../../../src/extras/react";


const slowlyAddTwoNumbers = (x: number, y: number, duration = 1000, name = 'unnamed') =>
  new Promise<number>((resolve) => setTimeout(() => resolve(x + y), duration));

const machine = withSubscribe(
  createPromiseMachine(
    slowlyAddTwoNumbers 
  )
)

export function ReactMachineDemo({}) {
  const [state] = useMachine(machine)
  return <div>
    <p>
      Action: {machine.getState().match({ // render based on state
        Idle: () => <>
        <button onClick={() => machine.send("execute", 1,1)}>
          Add 1+1
        </button>
      </>,
        Pending: ([x, y, duration, name]) => <span>
          Waiting {duration?.toString() ?? 'default=1000'}ms 
          to add {x} + {y} 
          {!!name && `(aka ${name})`}
        </span>,
        _: () => <>Done! <button onClick={machine.reset}>Reset</button></>
      })}
    </p>
    <pre>
      {JSON.stringify({
        "Current State Key": state.key,
        "Current State Data": state.data,
        "Last Change": machine.getChange()
      }, null, 2)}
    </pre>
  </div>;
}



```
